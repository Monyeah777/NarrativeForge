def scan_feeds(state):               # M57
    for npc in state.characters:

            feed = asset_roll("SOCIAL_FEED", npc=npc, state=state)
            state.player.social.feed_visible.append(feed)
            if feed.trigger_event:
                state.events.pending.append(feed.trigger_event)   # 动态作事件触发点
            if feed.clue_trail:
                state.hidden_events.clue_progress[npc.id] += 1    # 线索埋放

def scan_calls(state):               # M58
    for npc in state.characters:
        if asset_match("PHONE_CALL", npc=npc, events=state.events, affinity=state.player.affinity[npc.id]):
            call = asset_roll("PHONE_CALL", npc=npc, state=state)
            if call.affinity_probability and roll_probability(call.affinity_probability):
                state.player.social.phone_log.append(call)        # 来电

                    state.events.pending.append(call.trigger_event)  # 来电触发事件
